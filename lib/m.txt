REPOS=("Core-Architecture" "Data-Sync-Engine" "Auth-Protocol-v2" "UI-Component-Library" "Edge-Compute-Node" "Service-Mesh-Optimizer" "Legacy-Bridge")



START_DATE="2024-05-14"

END_DATE="2026-02-14"



MESSAGES=(

    "refactor: optimize data processing pipeline"

    "feat: implement asynchronous event handling"

    "fix: resolve memory leak in connection pool"

    "docs: update architecture design patterns"

    "perf: improve query execution time by 20%"

    "chore: dependency security patching"

    "style: standardized module export structure"

    "test: increase coverage for core utility functions"

    "feat: add telemetry hooks for service monitoring"

    "fix: race condition in distributed lock manager"

)



# ========== CLEANUP & PREP ==========

# Remove old failed attempts to avoid "Permission Denied"

rm -rf ~/github_stealth_work

mkdir -p ~/github_stealth_work && cd ~/github_stealth_work



# ========== EXECUTION ==========

for REPO_NAME in "${REPOS[@]}"; do

    echo "------------------------------------------------"

    echo "🏗️  Creating Repo: $REPO_NAME"

    

    # 1. Create the Repo via API (Check if it works)

    RESPONSE=$(curl -s -o /dev/null -w "%{http_code}" -H "Authorization: token $GITHUB_TOKEN" \

         -d "{\"name\":\"$REPO_NAME\", \"private\":true}" \

         https://api.github.com/user/repos)



    if [ "$RESPONSE" -ne 201 ] && [ "$RESPONSE" -ne 422 ]; then

        echo "❌ API Error: GitHub returned status $RESPONSE. Check your token permissions."

        continue

    fi

    

    # 2. Setup Local Repo

    echo "📥 Cloning $REPO_NAME..."

    git clone "https://$GITHUB_TOKEN@github.com/$USERNAME/$REPO_NAME.git" "$REPO_NAME" 2>/dev/null

    

    if [ ! -d "$REPO_NAME" ]; then

        echo "❌ Clone Error: Could not download $REPO_NAME. Skipping..."

        continue

    fi



    cd "$REPO_NAME"

    git config user.name "$USERNAME"

    git config user.email "$EMAIL"

    touch .build_metadata



    # 3. Generate History

    echo "⏳ Generating history..."

    curr="$START_DATE"

    total_repo_commits=0

    stop_date=$(date -I -d "$END_DATE + 1 day")



    while [ "$curr" != "$stop_date" ]; do

        dow=$(date -d "$curr" +%w) 

        intensity=$((RANDOM % 100))

        

        if [ "$intensity" -lt 5 ]; then

            num_commits=$((5 + RANDOM % 5)) # Crunch Day

            skip_chance=0 

        elif [ "$dow" -eq 0 ]; then

            num_commits=1 # Sunday Spike

            skip_chance=95

        elif [ "$dow" -eq 6 ]; then

            num_commits=$((1 + RANDOM % 2)) # Saturday

            skip_chance=85

        else

            num_commits=$((1 + RANDOM % 3)) # Normal

            skip_chance=82 

        fi



        if [ $((RANDOM % 100)) -lt $skip_chance ]; then

            curr=$(date -I -d "$curr + 1 day")

            continue

        fi



        day_start=$((7 + RANDOM % 8)) 

        day_window=$((6 + RANDOM % 6)) 



        for ((i=1; i<=$num_commits; i++)); do

            MSG=${MESSAGES[$RANDOM % ${#MESSAGES[@]}]}

            hour=$((day_start + RANDOM % day_window))

            [ $hour -ge 24 ] && hour=$((hour - 24)) 

            TIMESTAMP="$curr $(printf "%02d:%02d:%02d" $hour $((RANDOM % 60)) $((RANDOM % 60)))"



            echo "Log: $TIMESTAMP - $RANDOM" >> .build_metadata

            git add .build_metadata

            GIT_AUTHOR_DATE="$TIMESTAMP" GIT_COMMITTER_DATE="$TIMESTAMP" \

            git commit --date="$TIMESTAMP" -m "$MSG" --no-verify > /dev/null

            total_repo_commits=$((total_repo_commits + 1))

        done

        curr=$(date -I -d "$curr + 1 day")

    done



    # 4. Push and Cleanup

    echo "⬆️  Pushing $total_repo_commits commits..."

    git push origin main --force -q

    cd ..

    rm -rf "$REPO_NAME"

    echo "✅ $REPO_NAME Complete!"

done



echo "------------------------------------------------"

echo "🎉 ALL DONE!"
